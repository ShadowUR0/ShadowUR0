        if (id !== this.relicFx.id) {
            if (this.relicFx.group) {
                this.playerRoot.remove(this.relicFx.group);
                this.relicFx.group.traverse((o) => {
                    if (o instanceof THREE.Mesh) {
                        o.geometry.dispose();
                        o.material.dispose();
                    }
                });
            }
            if (this.relicFx.weapon)
                this.relicFx.weapon.mesh.material = this.relicFx.weapon.orig;
            this.relicFx = { id, group: null, weapon: null };
            if (id === 'mourning-edge') {
                let wm = null;
                this.playerRoot.traverse((o) => {
                    if (!wm && o instanceof THREE.Mesh && (o.name === '1H_Sword' || o.name === '2H_Staff'))
                        wm = o;
                });
                if (wm !== null) {
                    const mesh = wm;
                    const orig = mesh.material;
                    const m = (Array.isArray(orig) ? orig[0] : orig).clone();

                    m.emissiveIntensity = 2.2;
                    m.toneMapped = false;
                    mesh.material = m;

                    const light = new THREE.PointLight(0x6fe3d4, 5, 3.5);

            }
