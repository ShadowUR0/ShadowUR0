    `);
        const form = el.querySelector('#online-form');
        form.addEventListener('submit', (event) => {
            event.preventDefault();
            const join = el.querySelector('#btn-online-join');
            if (join?.disabled)
                return;
            if (join) {
                join.disabled = true;
                join.textContent = 'CONNECTING…';
            }
            const data = new FormData(form);
            this.onChoice({
                t: 'onlineJoin',
                room: String(data.get('room') ?? 'mere'),
                name: String(data.get('name') ?? 'Wanderer'),
                classId: data.get('classId') === 'vessel' ? 'vessel' : 'reaver',
                serverUrl: String(data.get('serverUrl') ?? defaultServer),
            });
        });
        el.querySelector('#btn-back')?.addEventListener('click', () => this.onChoice({ t: 'title' }));
        this.prepareDialog(el, 'online-dialog-title', 'input[name="name"]');
    }
