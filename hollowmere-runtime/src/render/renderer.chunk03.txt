    sync(sim, dtReal, timeMs, alpha = 1) {
        const t = timeMs / 1000;
        const SNAP = 3;
        const prev = this.interpPrev;
        const lerp1 = (a, b) => (Math.abs(b - a) > SNAP ? b : a + (b - a) * alpha);
        const ppx = prev ? lerp1(prev.px, sim.px) : sim.px;
        const ppz = prev ? lerp1(prev.pz, sim.pz) : sim.pz;
        this.playerRoot.position.set(ppx, 0, ppz);
        for (const view of this.remoteViews.values()) {
            const dx = view.tx - view.root.position.x, dz = view.tz - view.root.position.z;
            const snap = Math.hypot(dx, dz) > 8;
            const k = snap ? 1 : Math.min(1, dtReal * 12);
            view.root.position.x += dx * k;
            view.root.position.z += dz * k;
            view.avatar.setFacing(Math.atan2(view.faceX, view.faceZ));
            const movingRemote = Math.hypot(dx, dz) > 0.025;
            if (view.attack)
                view.avatar.play(view.classId === 'vessel' ? 'Spellcast_Shoot' : '1H_Melee_Attack_Slice_Diagonal', { once: true, fadeSec: 0.06 });
            else
                view.avatar.play(movingRemote ? 'Running_A' : 'Idle', { fadeSec: 0.1 });
            view.avatar.update(dtReal);
        }
        const face = Math.atan2(sim.faceX, sim.faceZ);
        this.playerAvatar?.setFacing(face);
        this.syncRelicVisual(sim, t);
        const dodging = sim.dodgeT >= 0;
        const sipping = sim.sipT >= 0;
        this.parryPoseT = Math.max(0, this.parryPoseT - dtReal);
        this.parrySuccessT = Math.max(0, this.parrySuccessT - dtReal);
        const parrying = sim.parryT >= 0 || this.parryPoseT > 0;
        const parrySuccess = this.parrySuccessT > 0;
        const moving = !!prev && Math.hypot(sim.px - prev.px, sim.pz - prev.pz) > 1e-4;
        if (this.playerAvatar) {
            const av = this.playerAvatar;
            if (sim.dead) {
                av.play('Death_A', { once: true, clampWhenFinished: true, fadeSec: 0.15 });
            }
            else if (dodging) {
                if (!this.prevDodging) {
                    const dur = av.clipDuration('Dodge_Forward');
                    av.play('Dodge_Forward', { once: true, fadeSec: 0.06, timeScale: dur / Math.max(0.05, PLAYER.dodgeTime) });
                }
            }
            else if (parrySuccess) {
                if (!this.prevParrySuccess) {
                    const dur = av.clipDuration('Block_Hit');
                    av.play('Block_Hit', { once: true, clampWhenFinished: true, fadeSec: 0.04, timeScale: dur / 0.38 });
                }
            }
            else if (parrying) {
                if (!this.prevParrying) {
                    const dur = av.clipDuration('Block');
                    av.play('Block', { once: true, clampWhenFinished: true, fadeSec: 0.05, timeScale: dur / 0.3 });
                }
            }
            else if (sim.atk !== 'idle') {
                if (this.prevAtk === 'idle') {
                    const clipName = sim.cls.melee ? '1H_Melee_Attack_Slice_Diagonal' : 'Spellcast_Shoot';
                    const total = sim.cls.windup + sim.cls.active + sim.cls.recover;
                    const dur = av.clipDuration(clipName);
                    av.play(clipName, { once: true, clampWhenFinished: true, fadeSec: 0.08, timeScale: dur / Math.max(0.05, total) });
                }
            }
            else if (sipping) {
                if (!this.prevSipping) {
                    const dur = av.clipDuration('Use_Item');
                    av.play('Use_Item', { once: true, clampWhenFinished: true, fadeSec: 0.1, timeScale: dur / Math.max(0.05, PLAYER.flaskSip) });
                }
            }
            else if (moving) {
                av.play('Running_A', { fadeSec: 0.15 });
            }
            else {
                const engaged = sim.enemies.some(en => en.state !== 'idle' && en.state !== 'dead');
                av.play(engaged && av.hasClip('Idle_Combat') ? 'Idle_Combat' : 'Idle', { fadeSec: 0.15 });
            }
            av.update(dtReal);
        }
        this.prevAtk = sim.atk;
        this.prevDodging = dodging;
        this.prevSipping = sipping;
        this.prevParrying = parrying;
        this.prevParrySuccess = parrySuccess;
        const seen = new Set();
        for (const e of sim.enemies) {
            seen.add(e.id);
            let v = this.enemyViews.get(e.id);
            if (!v) {
                v = this.makeEnemyView(e);
                this.enemyViews.set(e.id, v);
            }
            const pe = prev?.enemies.get(e.id);
            const ex = pe ? lerp1(pe.x, e.x) : e.x;
            const ez = pe ? lerp1(pe.z, e.z) : e.z;
            v.root.position.set(ex, 0, ez);
            v.avatar.setFacing(Math.atan2(e.faceX, e.faceZ));
            v.avatar.flash(e.hurtT > 0);
            if (e.state === 'dead') {
                v.avatar.play('Death_A', { once: true, clampWhenFinished: true, fadeSec: 0.15 });
                v.avatar.update(dtReal);
                v.deadT += dtReal;
                const k = Math.min(1, v.deadT / 0.9);
                v.root.scale.setScalar(1 - k * 0.85);
                v.root.position.y = -k * 0.9;
                v.telegraph.material.opacity = 0;
                v.telegraphArc.material.opacity = 0;
                v.telegraphFan.material.opacity = 0;
                v.telegraphRush.material.opacity = 0;
                v.shadow.visible = false;
                if (v.eliteRing)
                    v.eliteRing.visible = false;
                v.root.visible = k < 1;
                v.prevState = e.state;
                continue;
            }
            if (v.eliteRing) {
                v.eliteRing.position.set(ex, 0.085, ez);
                v.eliteRing.material.opacity = 0.24 + 0.08 * Math.sin(t * 2.6);
            }
            const enteringWindup = e.state === 'windup' && v.prevState !== 'windup';
            if (e.state === 'windup') {
                if (enteringWindup) {
                    const timing = enemyAttackTiming(e, sim);
                    const total = timing.windup + timing.active + timing.recover;
                    const clipName = enemyAttackClipName(e);
                    const dur = v.avatar.clipDuration(clipName);
                    v.avatar.play(clipName, { once: true, clampWhenFinished: true, fadeSec: 0.1, timeScale: dur / Math.max(0.05, total) });
                }
            }
            else if (e.state === 'active' || e.state === 'recover') {
            }
            else if (e.state === 'chase' || e.state === 'return') {
                v.avatar.play(e.kind === 'warden' ? 'Walking_A' : 'Running_A', { fadeSec: 0.15 });
            }
            else {
                v.avatar.play('Idle', { fadeSec: 0.15 });
            }
            v.avatar.update(dtReal);
            v.prevState = e.state;
            const tm = v.telegraph.material;
            const am = v.telegraphArc.material;
            const fm = v.telegraphFan.material;
            const rm = v.telegraphRush.material;
            tm.opacity = 0;
            am.opacity = 0;
            fm.opacity = 0;
            rm.opacity = 0;
            if (e.state === 'windup') {
                const k = Math.min(1, e.t / enemyAttackTiming(e, sim).windup);
                const pulse = 0.5 + 0.25 * Math.sin(t * 16) + 0.35 * k;
                if (e.attack === 'slam') {
                    v.telegraph.position.set(ex, 0.09, ez);
                    v.telegraph.scale.setScalar(BOSS.slamRadius);
                    tm.opacity = pulse * 0.8;
                }
                else if (e.attack === 'strike' || e.attack === 'swing') {
