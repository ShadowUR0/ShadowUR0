    showSkills(sim) {
        const cards = SLOTS.map(slot => {
            const def = skillFor(sim.cls.id, slot);
            const rank = sim.skillRanks[slot];
            const levelLocked = sim.level < def.requiredLevel;
            const canBuy = !levelLocked && sim.skillPoints > 0 && rank < def.maxRank;
            const status = levelLocked ? `UNLOCKS AT LEVEL ${def.requiredLevel}` : rank === 0 ? 'LOCKED · SPEND 1 POINT' : `RANK ${rank}/${def.maxRank}`;
            const damage = def.baseDamage + Math.max(0, rank - 1) * def.damagePerRank;
            return `<article class="skillcard ${rank === 0 ? 'locked' : ''}">
        <div class="skillmeta"><span>${SLOT_KEYS[slot]}</span><span>${status}</span></div>
        <h3>${def.name.toUpperCase()}</h3>
        <p>${def.description}</p>
        <div class="skillstats">${def.focusCost} FOCUS · ${def.cooldown.toFixed(0)}s BASE CD · ${damage} BASE DAMAGE</div>
        <button data-upgrade="${slot}" ${canBuy ? '' : 'disabled'}>${rank === 0 ? 'UNLOCK' : rank < def.maxRank ? 'UPGRADE' : 'MAX RANK'}</button>
      </article>`;
        }).join('');
        const el = this.openPanel(`
      <h2 id="skills-dialog-title">ABILITIES</h2>
      <div class="sub">${sim.cls.name.toUpperCase()} · LEVEL ${sim.level} · <strong>${sim.skillPoints} SKILL POINT${sim.skillPoints === 1 ? '' : 'S'}</strong></div>
      <div class="skillgrid">${cards}</div>
      <div class="btnrow"><button id="btn-resume">RETURN TO THE MERE</button></div>
    `, 'scrollpanel');
        el.querySelectorAll('[data-upgrade]').forEach(button => button.addEventListener('click', () => {

    }
