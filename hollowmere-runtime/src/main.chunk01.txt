function setupTouchControls() {
    const movezone = document.getElementById('touch-movezone');
    const joyBase = document.getElementById('tjoy-base');
    const joyKnob = document.getElementById('tjoy-knob');
    const positionJoy = (originX, originY, knobX, knobY) => {
        if (joyBase) {
            joyBase.style.left = `${originX}px`;
            joyBase.style.top = `${originY}px`;
        }
        if (joyKnob) {
            joyKnob.style.left = `${knobX}px`;
            joyKnob.style.top = `${knobY}px`;
        }
    };
    if (movezone) {
        movezone.addEventListener('pointerdown', (e) => {
            if (joyPointerId !== null)
                return;
            joyPointerId = e.pointerId;
            joyOriginX = e.clientX;
            joyOriginY = e.clientY;
            positionJoy(joyOriginX, joyOriginY, joyOriginX, joyOriginY);
            joyBase?.classList.add('on');
            joyKnob?.classList.add('on');
            movezone.setPointerCapture(e.pointerId);
            e.preventDefault();
        });
        movezone.addEventListener('pointermove', (e) => {
            if (e.pointerId !== joyPointerId)
                return;
            const dx = clamp(e.clientX - joyOriginX, -JOY_RADIUS, JOY_RADIUS);
            const dy = clamp(e.clientY - joyOriginY, -JOY_RADIUS, JOY_RADIUS);
            positionJoy(joyOriginX, joyOriginY, joyOriginX + dx, joyOriginY + dy);
            const rightAmt = dx / JOY_RADIUS, upAmt = -dy / JOY_RADIUS;
            touchMoveX = upAmt * F[0] + rightAmt * R[0];
            touchMoveZ = upAmt * F[1] + rightAmt * R[1];
            e.preventDefault();
        });
        const endJoy = (e) => {
            if (e.pointerId !== joyPointerId)
                return;
            joyPointerId = null;
            touchMoveX = 0;
            touchMoveZ = 0;
            joyBase?.classList.remove('on');
            joyKnob?.classList.remove('on');
        };
        movezone.addEventListener('pointerup', endJoy);
        movezone.addEventListener('pointercancel', endJoy);
    }
    const bindPulse = (id, set) => {
        const el = document.getElementById(id);
        if (!el)
            return;
        el.addEventListener('pointerdown', (e) => {
            set();
            el.classList.add('pressed');
            el.setPointerCapture(e.pointerId);
            e.preventDefault();
        });
        const release = () => el.classList.remove('pressed');
        el.addEventListener('pointerup', release);
        el.addEventListener('pointercancel', release);
    };
    bindPulse('tbtn-skill1', () => { pressed.skill = true; });
    bindPulse('tbtn-skill2', () => { pressed.skill2 = true; });
    bindPulse('tbtn-skill3', () => { pressed.skill3 = true; });
    bindPulse('tbtn-ultimate', () => { pressed.ultimate = true; });
    bindPulse('tbtn-parry', () => { pressed.parry = true; });
    bindPulse('tbtn-dodge', () => { pressed.dodge = true; });
    bindPulse('tbtn-flask', () => { pressed.flask = true; });
    bindPulse('tbtn-interact', () => { pressed.interact = true; });
    const attackBtn = document.getElementById('tbtn-attack');
    if (attackBtn) {
        attackBtn.addEventListener('pointerdown', (e) => {
            touchAttackHeld = true;
            aimPointerId = e.pointerId;
            attackBtn.classList.add('pressed');
            attackBtn.setPointerCapture(e.pointerId);
            e.preventDefault();
        });
        attackBtn.addEventListener('pointermove', (e) => {
            if (e.pointerId !== aimPointerId)
                return;
            const rect = attackBtn.getBoundingClientRect();
            const cx = rect.left + rect.width / 2, cy = rect.top + rect.height / 2;
            const dx = e.clientX - cx, dy = e.clientY - cy;
            if (Math.hypot(dx, dy) < AIM_DEADZONE) {
                touchAimActive = false;
                touchAimX = 0;
                touchAimZ = 0;
                return;
            }
            const rightAmt = clamp(dx / AIM_RADIUS, -1, 1), upAmt = clamp(-dy / AIM_RADIUS, -1, 1);
            touchAimActive = true;
            touchAimX = upAmt * F[0] + rightAmt * R[0];
            touchAimZ = upAmt * F[1] + rightAmt * R[1];
            e.preventDefault();
        });
        const endAttack = (e) => {
            if (e.pointerId !== aimPointerId)
                return;
            aimPointerId = null;
            touchAttackHeld = false;
            touchAimActive = false;
            touchAimX = 0;
            touchAimZ = 0;
            attackBtn.classList.remove('pressed');
        };
        attackBtn.addEventListener('pointerup', endAttack);
        attackBtn.addEventListener('pointercancel', endAttack);
    }
    const dispatchKey = (code) => {
        dispatchEvent(new KeyboardEvent('keydown', { code, bubbles: true, cancelable: true }));
    };
    document.getElementById('tbtn-pause')?.addEventListener('click', () => dispatchKey('Escape'));
    document.getElementById('tbtn-inventory')?.addEventListener('click', () => dispatchKey('Tab'));
    document.getElementById('tbtn-skills')?.addEventListener('click', () => dispatchKey('KeyK'));
}

function gatherInput(s) {
    if (BOT)
        return botInput(s);
    let ix = 0, iz = 0;
    if (keys.has('KeyW') || keys.has('ArrowUp')) {
        ix += F[0];
        iz += F[1];
    }
    if (keys.has('KeyS') || keys.has('ArrowDown')) {
        ix -= F[0];
        iz -= F[1];
    }
    if (keys.has('KeyD') || keys.has('ArrowRight')) {
        ix += R[0];
        iz += R[1];
    }
    if (keys.has('KeyA') || keys.has('ArrowLeft')) {
        ix -= R[0];
        iz -= R[1];
    }
    if (TOUCH_ACTIVE) {
        ix += touchMoveX;
        iz += touchMoveZ;
    }
    let aimX = 0, aimZ = 0;
    if (TOUCH_ACTIVE) {
        if (touchAimActive) {
            aimX = touchAimX;
            aimZ = touchAimZ;
        }
    }
    else {
        const g = renderer.groundPoint(mouseX, mouseY);
        aimX = g ? g.x - s.px : 0;
        aimZ = g ? g.z - s.pz : 0;
    }
    const inp = {
        mx: ix, mz: iz,
        aimX, aimZ,
        attack: lmb || touchAttackHeld,
        skill: pressed.skill, skill2: pressed.skill2, skill3: pressed.skill3, ultimate: pressed.ultimate, parry: pressed.parry,
        dodge: pressed.dodge, flask: pressed.flask, interact: pressed.interact,
    };
    pressed.skill = pressed.skill2 = pressed.skill3 = pressed.ultimate = pressed.parry = pressed.dodge = pressed.flask = pressed.interact = false;
    return inp;
}
